  # Convert the input arguments into a single regex group (e.g., TestFoo|TestBar)
  local regex_group=""
  for test_name in "${test_names[@]}"; do
    if [ -z "$regex_group" ]; then
      regex_group="$test_name"

  regex_group="^($regex_group)$"

        'SKIP': TestStatus.SKIPPED,
    }

    results: List[TestResult] = []
    # Go's `go test -v` indents sub-test result lines (t.Run) with leading
    # whitespace, so allow it; capture PASS/FAIL/SKIP, not just PASS.
    pattern = re.compile(r'^\s*--- (PASS|FAIL|SKIP):\s+(\S+)\s+\(.*\)$')

    for line in stdout_content.splitlines():

        if not m:
            continue
        key, name = m.groups()
        status = status_map.get(key)
        if status:
            results.append(TestResult(name=name, status=status))

    return results
